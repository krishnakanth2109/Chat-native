// ── Voice Player ──────────────────────────────────────────────
const VoicePlayer = ({ url, isMe }: { url: string; isMe: boolean }) => {
  const [sound, setSound] = useState<Audio.Sound | null>(null);
  const [isPlaying, setIsPlaying] = useState(false);
  const [duration, setDuration] = useState(0);
  const [position, setPosition] = useState(0);
  const { colors } = useAppTheme();

  const handlePlayPause = async () => {
    if (!sound) {
      const { sound: ns } = await Audio.Sound.createAsync({ uri: url }, {}, (s: any) => {
        if (s.isLoaded) { setDuration(s.durationMillis || 0); setPosition(s.positionMillis || 0); }
        if (s.didJustFinish) setIsPlaying(false);
      });
      setSound(ns); await ns.playAsync(); setIsPlaying(true);
    } else {
      if (isPlaying) { await sound.pauseAsync(); setIsPlaying(false); }
      else { await sound.playAsync(); setIsPlaying(true); }
    }
  };
  useEffect(() => { return () => { if (sound) sound.unloadAsync(); }; }, [sound]);
  const progress = duration > 0 ? position / duration : 0;
  const fmt = (ms: number) => `${Math.floor(ms / 60000)}:${String(Math.floor((ms % 60000) / 1000)).padStart(2, '0')}`;

  return (
    <View style={{ flexDirection: 'row', alignItems: 'center', gap: 10, minWidth: 180, paddingVertical: 2 }}>
      <TouchableOpacity onPress={handlePlayPause} style={{ backgroundColor: isMe ? 'rgba(255,255,255,0.25)' : 'rgba(0,0,0,0.1)', borderRadius: 20, padding: 6 }}>
        <Ionicons name={isPlaying ? 'pause' : 'play'} size={20} color={isMe ? '#fff' : colors.primary} />
      </TouchableOpacity>
      <View style={{ flex: 1 }}>
        <View style={{ height: 3, backgroundColor: isMe ? 'rgba(255,255,255,0.3)' : colors.border, borderRadius: 2 }}>
          <View style={{ width: `${progress * 100}%`, height: '100%', backgroundColor: isMe ? '#fff' : colors.primary, borderRadius: 2 }} />
        </View>
        <Text style={{ fontSize: 10, color: isMe ? 'rgba(255,255,255,0.7)' : colors.textSecondary, marginTop: 3 }}>{fmt(position)} / {fmt(duration)}</Text>
      </View>
      <Ionicons name="mic" size={13} color={isMe ? 'rgba(255,255,255,0.7)' : colors.textSecondary} />
    </View>
  );
};

// ── Chat Input Bar ────────────────────────────────────────────
const ChatInputBar = React.memo(({
  onSend, onSendEmoji, onSendSticker, sending, showAttach, setShowAttach, pickImage, sendLocation, sendVoiceMessage,
}: {
  onSend: (t: string) => void; onSendEmoji: (e: string) => void; onSendSticker: (s: string) => void;
  sending: boolean; showAttach: boolean; setShowAttach: React.Dispatch<React.SetStateAction<boolean>>;
  pickImage: (cam: boolean) => void; sendLocation: () => void; sendVoiceMessage: (uri: string) => void;
}) => {
  const [text, setText] = useState('');
  const [recording, setRecording] = useState<Audio.Recording | null>(null);
  const [showEmoji, setShowEmoji] = useState(false);
  const { colors } = useAppTheme();
  const s = React.useMemo(() => getStyles(colors), [colors]);

  const handleSend = () => {
    if (!text.trim()) return;
    onSend(text.trim()); setText(''); setShowAttach(false); setShowEmoji(false);
  };
  const startRecording = async () => {
    try {
      const perm = await Audio.requestPermissionsAsync();
      if (perm.status !== 'granted') return Alert.alert('Permission denied', 'Microphone access is required.');
      await Audio.setAudioModeAsync({ allowsRecordingIOS: true, playsInSilentModeIOS: true });
      const { recording: rec } = await Audio.Recording.createAsync(Audio.RecordingOptionsPresets.HIGH_QUALITY);
      setRecording(rec);
    } catch { Alert.alert('Error', 'Failed to start recording'); }
  };
  const stopRecording = async () => {
    if (!recording) return;
    try {
      await recording.stopAndUnloadAsync();
      const uri = recording.getURI(); setRecording(null);
      if (uri) sendVoiceMessage(uri);
    } catch { Alert.alert('Error', 'Failed to stop recording'); }
  };

  return (
    <View>
      {showEmoji && <EmojiPicker onSelectEmoji={(e) => { setText(prev => prev + e); }} onSelectSticker={(s) => { onSendSticker(s); setShowEmoji(false); }} />}
      <View style={s.inputBar}>
        <TouchableOpacity onPress={() => { setShowEmoji(v => !v); setShowAttach(false); }} style={s.addBtn}>
          <Text style={{ fontSize: 27 }}>{showEmoji ? '⌨️' : '😊'}</Text>
        </TouchableOpacity>
        <View style={s.inputPill}>
          <TextInput style={s.input} value={text} onChangeText={setText} placeholder="Message…" placeholderTextColor={colors.textSecondary} multiline maxLength={1000}
            onFocus={() => { setShowAttach(false); setShowEmoji(false); }} />
          {text.trim() ? (
            sending ? <ActivityIndicator size="small" color={colors.primary} style={{ paddingHorizontal: 10 }} /> : (
              <TouchableOpacity onPress={handleSend} style={s.sendBtn}>
                <View style={[s.sendBtnInner, { backgroundColor: colors.primary }]}>
                  <Ionicons name="send" size={17} color="#fff" />
                </View>
              </TouchableOpacity>
            )
          ) : (
            <View style={s.rightIcons}>
              <TouchableOpacity onPressIn={startRecording} onPressOut={stopRecording} style={[s.iconBtn, recording && { backgroundColor: '#ef444433' }]}>
                <Ionicons name={recording ? 'radio-button-on' : 'mic-outline'} size={21} color={recording ? '#ef4444' : colors.textSecondary} />
              </TouchableOpacity>
              <TouchableOpacity onPress={() => { setShowAttach(v => !v); setShowEmoji(false); }} style={s.iconBtn}>
                <Ionicons name="attach-outline" size={21} color={colors.textSecondary} />
              </TouchableOpacity>
            </View>
          )}
        </View>
      </View>
    </View>
  );
});
ChatInputBar.displayName = 'ChatInputBar';

// ── MAIN SCREEN ───────────────────────────────────────────────
export default function ChatScreen() {
  const params = useLocalSearchParams<{ id: string; name: string; avatar: string }>();
  const id = Array.isArray(params.id) ? params.id[0] : params.id;
  const name = Array.isArray(params.name) ? params.name[0] : params.name;
  const avatar = Array.isArray(params.avatar) ? params.avatar[0] : (params.avatar || '');
  const { userId } = useAuth();
  const callContext = React.useContext(CallContext);
  const { colors } = useAppTheme();
  const s = React.useMemo(() => getStyles(colors), [colors]);
  const [messages, setMessages] = useState<Message[]>([]);
  const [loading, setLoading] = useState(true);
  const [sending, setSending] = useState(false);
  const [isOnline, setIsOnline] = useState(false);
  const [showAttach, setShowAttach] = useState(false);
  const [isSocketConnected, setIsSocketConnected] = useState(socket.connected);
  const [page, setPage] = useState(1);
  const [hasMore, setHasMore] = useState(true);
  const listRef = useRef<FlatList>(null);

  const loadMessages = useCallback(async (pageNum: number = 1) => {
    if (!id) return;
    try {
      if (pageNum === 1) setLoading(true);
      const { data } = await api.get<Message[]>(`/messages/${id}?page=${pageNum}&limit=30`);
      if (data.length < 30) setHasMore(false);
      setMessages(prev => pageNum === 1 ? data : [...prev, ...data]);
      setPage(pageNum);
    } catch (err: any) { console.error(err?.message); }
    finally { if (pageNum === 1) setLoading(false); }
  }, [id]);
  useEffect(() => { loadMessages(1); }, [loadMessages]);

  useEffect(() => {
    const onMsg = (msg: any) => {
      const sid = typeof msg.sender === 'object' ? msg.sender._id : msg.sender;
      const rid = typeof msg.receiver === 'object' ? msg.receiver._id : msg.receiver;
      if ((sid === id && rid === userId) || (sid === userId && rid === id)) {
        setMessages(prev => {
          const exists = prev.find(m => (m._id === msg._id || (m.content === msg.content && m.status === 'pending')) && m.sender === sid);
          if (exists) return prev.map(m => m._id === exists._id ? { ...m, ...msg, status: 'sent' } : m);
          return [{ ...msg, sender: sid, receiver: rid, status: 'sent' }, ...prev];
        });
      }
    };
    const onOnline = (ids: string[]) => setIsOnline(ids.includes(id ?? ''));
    const onConnect = () => setIsSocketConnected(true);
    const onDisconnect = () => setIsSocketConnected(false);
    socket.on('receive_message', onMsg); socket.on('getOnlineUsers', onOnline);
    socket.on('connect', onConnect); socket.on('disconnect', onDisconnect);
    setIsSocketConnected(socket.connected);
    return () => {
      socket.off('receive_message', onMsg); socket.off('getOnlineUsers', onOnline);
      socket.off('connect', onConnect); socket.off('disconnect', onDisconnect);
    };
  }, [id, userId]);

  const emitMessage = (payload: any, cid: string) => {
    if (!socket.connected && userId) connectSocket(userId);
    socket.emit('send_message', payload, (r: any) => {
      if (r?.status === 'ok') setMessages(prev => prev.map(m => m._id === cid ? { ...r.data, status: 'sent' } : m));
      else setMessages(prev => prev.map(m => m._id === cid ? { ...m, status: 'error' } : m));
    });
  };

  const sendText = useCallback((t: string) => {
    if (!t || !userId) return;
    const cid = Date.now().toString() + Math.random().toString(36).slice(2);
    setMessages(prev => [{ _id: cid, sender: userId, receiver: id, messageType: 'text', content: t, createdAt: new Date().toISOString(), status: 'pending' }, ...prev]);
    emitMessage({ sender: userId, receiver: id, messageType: 'text', content: t }, cid);
  }, [userId, id]);

  const sendEmoji = useCallback((e: string) => {
    if (!userId) return;
    const cid = Date.now().toString() + Math.random().toString(36).slice(2);
    setMessages(prev => [{ _id: cid, sender: userId, receiver: id, messageType: 'emoji', content: e, createdAt: new Date().toISOString(), status: 'pending' }, ...prev]);
    emitMessage({ sender: userId, receiver: id, messageType: 'text', content: e }, cid);
  }, [userId, id]);

  const sendSticker = useCallback((st: string) => {
    if (!userId) return;
    const cid = Date.now().toString() + Math.random().toString(36).slice(2);
    setMessages(prev => [{ _id: cid, sender: userId, receiver: id, messageType: 'sticker', content: st, createdAt: new Date().toISOString(), status: 'pending' }, ...prev]);
    emitMessage({ sender: userId, receiver: id, messageType: 'text', content: st }, cid);
  }, [userId, id]);

  const uploadFile = async (uri: string, type: 'image' | 'document', fileName?: string) => {
    const ext = uri.split('.').pop() || 'bin';
    const mime = type === 'image' ? `image/${ext === 'jpg' ? 'jpeg' : ext}` : 'application/octet-stream';
    const form = new FormData();
    form.append('media', { uri, type: mime, name: fileName || `file.${ext}` } as any);
    const { data } = await api.post('/upload/media', form, { headers: { 'Content-Type': 'multipart/form-data' } });
    if (!data.filePath) throw new Error('No filePath');
    return data.filePath as string;
  };

  const pickImage = async (useCamera = false) => {
    setShowAttach(false); let cid = '';
    try {
      let res: ImagePicker.ImagePickerResult;
      if (useCamera) {
        const p = await ImagePicker.requestCameraPermissionsAsync();
        if (!p.granted) return Alert.alert('Permission required');
        res = await ImagePicker.launchCameraAsync({ quality: 0.8 });
      } else { res = await ImagePicker.launchImageLibraryAsync({ quality: 0.8 }); }
      if (res.canceled) return;
      const localUri = res.assets[0].uri;
      cid = Date.now().toString() + Math.random().toString(36).slice(2);
      setMessages(prev => [{ _id: cid, sender: userId!, receiver: id, messageType: 'image', fileUrl: localUri, createdAt: new Date().toISOString(), status: 'pending' }, ...prev]);
      setSending(true);
      const url = await uploadFile(localUri, 'image');
      emitMessage({ sender: userId, receiver: id, messageType: 'image', fileUrl: url }, cid);
    } catch {
      if (cid) setMessages(prev => prev.map(m => m._id === cid ? { ...m, status: 'error' } : m));
      Alert.alert('Error', 'Failed to send image.');
    } finally { setSending(false); }
  };

  const pickDocument = async () => {
    setShowAttach(false); let cid = '';
    try {
      const res = await DocumentPicker.getDocumentAsync({ type: '*/*' });
      if (res.canceled) return;
      const file = res.assets[0];
      cid = Date.now().toString() + Math.random().toString(36).slice(2);
      setMessages(prev => [{ _id: cid, sender: userId!, receiver: id, messageType: 'document', fileName: file.name, createdAt: new Date().toISOString(), status: 'pending' }, ...prev]);
      setSending(true);
      const url = await uploadFile(file.uri, 'document', file.name);
      emitMessage({ sender: userId, receiver: id, messageType: 'document', fileUrl: url, fileName: file.name }, cid);
    } catch {
      if (cid) setMessages(prev => prev.map(m => m._id === cid ? { ...m, status: 'error' } : m));
      Alert.alert('Error', 'Failed to send document.');
    } finally { setSending(false); }
  };

  const sendLocation = async () => {
    setShowAttach(false);
    const { status } = await Location.requestForegroundPermissionsAsync();
    if (status !== 'granted') return Alert.alert('Permission denied', 'Enable location access.');
    try {
      setSending(true);
      let loc = await Location.getLastKnownPositionAsync({});
      if (!loc) loc = await Location.getCurrentPositionAsync({ accuracy: Location.Accuracy.Balanced });
      if (!loc) return;
      let address = '';
      try {
        const geo = await Location.reverseGeocodeAsync({ latitude: loc.coords.latitude, longitude: loc.coords.longitude });
        if (geo.length > 0) {
          const g = geo[0];
          address = [g.name, g.district, g.city, g.region].filter(Boolean).slice(0, 3).join(', ');
        }
      } catch { /* fallback to coords */ }
      const cid = Date.now().toString() + Math.random().toString(36).slice(2);
      setMessages(prev => [{ _id: cid, sender: userId!, receiver: id, messageType: 'location', location: { lat: loc!.coords.latitude, lng: loc!.coords.longitude, address }, createdAt: new Date().toISOString(), status: 'pending' }, ...prev]);
      socket.emit('send_message', { sender: userId, receiver: id, messageType: 'location', location: { lat: loc.coords.latitude, lng: loc.coords.longitude, address } }, (r: any) => {
        if (r?.status === 'ok') setMessages(prev => prev.map(m => m._id === cid ? { ...r.data, status: 'sent' } : m));
        else setMessages(prev => prev.map(m => m._id === cid ? { ...m, status: 'error' } : m));
      });
    } catch (err: any) { Alert.alert('Error', err?.message || 'Could not get location.'); }
    finally { setSending(false); }
  };

  const sendVoiceMessage = async (uri: string) => {
    const cid = Date.now().toString() + Math.random().toString(36).slice(2);
    try {
      setMessages(prev => [{ _id: cid, sender: userId!, receiver: id, messageType: 'voice', fileUrl: uri, createdAt: new Date().toISOString(), status: 'pending' }, ...prev]);
      setSending(true);
      const url = await uploadFile(uri, 'document', 'voice.m4a');
      emitMessage({ sender: userId, receiver: id, messageType: 'voice', fileUrl: url }, cid);
    } catch {
      setMessages(prev => prev.map(m => m._id === cid ? { ...m, status: 'error' } : m));
      Alert.alert('Error', 'Failed to send voice message.');
    } finally { setSending(false); }
  };

  const displayName = name || 'Chat';

  return (
    <View style={s.root}>
      <StatusBar barStyle="light-content" backgroundColor={colors.headerBackground} />
      <Stack.Screen options={{ headerShown: false }} />
      <SafeAreaView style={s.headerSafe}>
        <View style={s.header}>
          <TouchableOpacity onPress={() => router.back()} style={s.headerBack}>
            <Ionicons name="arrow-back" size={24} color={colors.headerIcon} />
          </TouchableOpacity>
          <View style={s.headerAvatar}>
            {avatar ? <Image source={{ uri: avatar }} style={{ width: 42, height: 42, borderRadius: 21 }} /> : <Text style={s.headerAvatarText}>{displayName.charAt(0).toUpperCase()}</Text>}
            {isOnline && <View style={s.onlineDot} />}
          </View>
          <View style={s.headerInfo}>
            <Text style={s.headerName} numberOfLines={1}>{displayName}</Text>
            <View style={{ flexDirection: 'row', alignItems: 'center', gap: 4 }}>
              {isOnline && <View style={{ width: 6, height: 6, borderRadius: 3, backgroundColor: colors.success }} />}
              <Text style={[s.headerStatus, { color: isOnline ? colors.success : colors.textSecondary }]}>{isOnline ? 'Active now' : 'Offline'}</Text>
            </View>
          </View>
          <View style={s.headerActions}>
            <TouchableOpacity onPress={() => callContext?.startCall(id, false)} style={s.headerIcon}>
              <Ionicons name="call-outline" size={21} color={colors.headerIcon} />
            </TouchableOpacity>
            <TouchableOpacity onPress={() => callContext?.startCall(id, true)} style={s.headerIcon}>
              <Ionicons name="videocam-outline" size={22} color={colors.headerIcon} />
            </TouchableOpacity>
          </View>
        </View>
        {!isSocketConnected && <View style={{ backgroundColor: colors.error, paddingVertical: 4, alignItems: 'center' }}><Text style={{ color: '#fff', fontSize: 11, fontWeight: '700' }}>Reconnecting…</Text></View>}
      </SafeAreaView>

      <KeyboardAvoidingView style={{ flex: 1 }} behavior={Platform.OS === 'ios' ? 'padding' : 'height'} keyboardVerticalOffset={0}>
        {loading ? (
          <View style={s.center}>
            <ActivityIndicator size="large" color={colors.primary} />
            <Text style={{ color: colors.textSecondary, marginTop: 10, fontSize: 13 }}>Loading messages…</Text>
          </View>
        ) : (
          <FlatList ref={listRef} data={messages} inverted keyExtractor={item => item._id}
            renderItem={({ item }) => { const sid = typeof item.sender === 'object' ? (item.sender as any)._id : item.sender; return <Bubble item={item} isMe={sid === userId} />; }}
            contentContainerStyle={{ padding: 12, paddingBottom: 8, flexGrow: 1 }}
            onEndReached={() => { if (hasMore) loadMessages(page + 1); }}
            onEndReachedThreshold={0.5} initialNumToRender={15} maxToRenderPerBatch={10} windowSize={5}
            removeClippedSubviews={Platform.OS === 'android'}
            ListEmptyComponent={
              <View style={s.empty}>
                <Text style={{ fontSize: 54 }}>👋</Text>
                <Text style={s.emptyTxt}>Say hello to {displayName}!</Text>
                <Text style={{ color: colors.textSecondary, fontSize: 13, marginTop: 4 }}>Start a conversation</Text>
              </View>
            }
          />
        )}
        {showAttach && (
          <View style={s.attachTray}>
            {[
              { icon: 'camera', label: 'Camera', color: '#6366f1', action: () => pickImage(true) },
              { icon: 'images', label: 'Gallery', color: '#0ea5e9', action: () => pickImage(false) },
              { icon: 'document-text', label: 'Document', color: '#f59e0b', action: pickDocument },
              { icon: 'location', label: 'Location', color: '#ef4444', action: sendLocation },
            ].map(btn => (
              <TouchableOpacity key={btn.label} style={s.attachItem} onPress={btn.action}>
                <View style={[s.attachIcon, { backgroundColor: btn.color }]}>
                  <Ionicons name={btn.icon as any} size={22} color="#fff" />
                </View>
                <Text style={s.attachLabel}>{btn.label}</Text>
              </TouchableOpacity>
            ))}
          </View>
        )}
        <ChatInputBar onSend={sendText} onSendEmoji={sendEmoji} onSendSticker={sendSticker} sending={sending}
          showAttach={showAttach} setShowAttach={setShowAttach} pickImage={pickImage}
          sendLocation={sendLocation} sendVoiceMessage={sendVoiceMessage} />
      </KeyboardAvoidingView>
    </View>
  );
}
